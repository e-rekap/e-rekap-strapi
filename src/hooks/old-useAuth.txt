import { createContext, useContext, useState, type ReactNode } from "react";
import type { AppUser } from "models/user";
import { DEV_USERS } from "variables/devUsers";

type AuthValue = { user: AppUser; setUser: (u: AppUser) => void };

// 1. Bikin "saluran"-nya
const AuthContext = createContext<AuthValue | null>(null);

// simpen user ke local storage biar kalo refresh ga ilang
const STORAGE_KEY = "erekap/devUser";
function loadUser(): AppUser {
  try {
    const id = localStorage.getItem(STORAGE_KEY);
    return DEV_USERS.find((u) => u.id === id) ?? DEV_USERS[0];
  } catch {
    return DEV_USERS[0]; // localStorage bisa error, misalnya di mode private
  }
}

// 2. Provider: yang ngirim data ke saluran
export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUserState] = useState<AppUser>(loadUser); // kasih FUNGSI-nya, jangan hasilnya

  const setUser = (u: AppUser) => {
    setUserState(u);
    try {
      localStorage.setItem(STORAGE_KEY, u.id);
    } catch {}
  };
  return (
    <AuthContext.Provider value={{ user, setUser }}>
      {children}
    </AuthContext.Provider>
  );
}

// 3. Hook: cara komponen ngambil data dari saluran
export function useAuth() {
  const value = useContext(AuthContext);
  if (!value) throw new Error("useAuth harus dipakai di dalam <AuthProvider>");
  return value;
}
